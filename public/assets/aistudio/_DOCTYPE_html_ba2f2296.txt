<script>
let USER = null;
let USERNAME = null;
let ROLE = null;
let ID_ANGGOTA = null;

function byId(id) {
  return document.getElementById(id);
}

function notifSukses(pesan) {
  Swal.fire({ icon: "success", title: "Berhasil", text: pesan, timer: 1800, showConfirmButton: false });
}

function notifError(pesan) {
  Swal.fire({ icon: "error", title: "Gagal", text: pesan });
}

function rupiah(n) {
  return "Rp " + Number(n || 0).toLocaleString("id-ID");
}

function showPage(page) {
  document.querySelectorAll(".page").forEach(p => p.classList.add("d-none"));

  if (page === "dashboard") {
    byId("pageDashboard").classList.remove("d-none");
    loadDashboard();
    loadRingkasanAnggota();
  }

  if (page === "anggota") {
    byId("pageAnggota").classList.remove("d-none");
    loadAnggota();
  }

  if (page === "kasMasuk") {
    byId("pageKasMasuk").classList.remove("d-none");
    loadPilihanAnggota();
  }

  if (page === "kasKeluar") {
    byId("pageKasKeluar").classList.remove("d-none");
  }

  if (page === "pinjaman") {
    byId("pagePinjaman").classList.remove("d-none");
    loadPilihanAnggota();
  }

  if (page === "angsuran") {
    byId("pageAngsuran").classList.remove("d-none");
    loadPilihanPinjaman();
  }

  if (page === "tunggakan") {
    byId("pageTunggakan").classList.remove("d-none");
    loadTunggakan();
  }

  if (page === "anggotaSaya") {
    byId("pageAnggotaSaya").classList.remove("d-none");
    loadDashboardAnggota();
  }

  if (page === "ajukanPinjaman") {
    byId("pageAjukanPinjaman").classList.remove("d-none");
  }

  if (page === "pengajuanPinjaman") {
    byId("pagePengajuanPinjaman").classList.remove("d-none");
    loadPengajuanPinjaman();
  }

  if (page === "pengaturan") {
    byId("pagePengaturan").classList.remove("d-none");
    byId("usernameBaru").value = USERNAME;
  }

  if (page === "export") {
    byId("pageExport").classList.remove("d-none");
  }
}

function login() {
  const username = byId("username").value;
  const password = byId("password").value;

  google.script.run
    .withSuccessHandler(res => {
      if (!res.status) {
        notifError(res.message);
        return;
      }

      USER = res.nama;
      USERNAME = res.username;
      ROLE = res.role;
      ID_ANGGOTA = res.idAnggota;

      byId("userInfo").innerText = res.nama + " (" + res.role + ")";
      byId("sideUserName").innerText = res.nama;
      byId("sideUserRole").innerText = res.role;

      byId("loginBox").style.display = "none";
      byId("daftarBox").style.display = "none";
      byId("app").style.display = "block";

      setMenuByRole(ROLE);
    })
    .withFailureHandler(err => notifError(err.message))
    .login(username, password);
}

function setMenuByRole(role) {
  const menuIds = [
    "menuDashboard", "menuAnggota", "menuKasMasuk", "menuKasKeluar",
    "menuPinjaman", "menuAngsuran", "menuTunggakan", "menuAnggotaSaya",
    "menuAjukanPinjaman", "menuPengajuanPinjaman", "menuPengaturan", "menuExport"
  ];

  menuIds.forEach(id => byId(id).style.display = "none");

  if (role === "Administrator") {
    ["menuDashboard", "menuAnggota", "menuKasMasuk", "menuKasKeluar", "menuPinjaman", "menuAngsuran", "menuTunggakan", "menuPengajuanPinjaman", "menuPengaturan", "menuExport"]
      .forEach(id => byId(id).style.display = "block");
    showPage("dashboard");
    return;
  }

  if (role === "Bendahara") {
    ["menuKasMasuk", "menuKasKeluar", "menuPinjaman", "menuAngsuran", "menuTunggakan", "menuPengaturan"]
      .forEach(id => byId(id).style.display = "block");
    showPage("kasMasuk");
    return;
  }

  if (role === "Anggota") {
    ["menuAnggotaSaya", "menuAjukanPinjaman", "menuPengaturan"]
      .forEach(id => byId(id).style.display = "block");
    showPage("anggotaSaya");
    return;
  }

  notifError("Role tidak dikenal: " + role);
}

function logout() {
  USER = null;
  USERNAME = null;
  ROLE = null;
  ID_ANGGOTA = null;

  byId("app").style.display = "none";
  byId("daftarBox").style.display = "none";
  byId("loginBox").style.display = "block";

  byId("username").value = "";
  byId("password").value = "";

  document.querySelectorAll(".page").forEach(p => p.classList.add("d-none"));
  window.scrollTo(0, 0);
}

      <div class="col-md-3 mb-3"><div class="card shadow-sm p-3 stat-card"><small>Total Anggota</small><h4>${d.anggota}</h4></div></div>
      <div class="col-md-3 mb-3"><div class="card shadow-sm p-3 stat-card"><small>Saldo Kas Kelompok</small><h4>${rupiah(d.saldo)}</h4></div></div>
      <div class="col-md-3 mb-3"><div class="card shadow-sm p-3 stat-card"><small>Total Kas Masuk</small><h4>${rupiah(d.masuk)}</h4></div></div>
      <div class="col-md-3 mb-3"><div class="card shadow-sm p-3 stat-card"><small>Total Piutang</small><h4>${rupiah(d.piutang)}</h4></div></div>
    `;
  }).withFailureHandler(err => notifError(err.message)).getDashboard();
}

function formatNoWA(nohp) {
  let nomor = String(nohp || "").replace(/\D/g, "");

  if (nomor.startsWith("0")) {
    nomor = "62" + nomor.substring(1);
  }

  if (!nomor.startsWith("62")) {
    nomor = "62" + nomor;
  }

  return nomor;
}

function kirimWAAngsuran(nohp, nama, idPinjaman, angsuranPerBulan, sisa) {
  const nomor = formatNoWA(nohp);

  if (!nomor || nomor.length < 10) {
    notifError("Nomor WhatsApp anggota belum valid");
    return;
  }

  const pesan =
    `Assalamu'alaikum ${nama},%0A%0A` +
    `Kami informasikan bahwa angsuran pinjaman Anda masih berjalan.%0A%0A` +
    `ID Pinjaman: ${idPinjaman}%0A` +
    `Angsuran yang harus dibayar: ${rupiah(angsuranPerBulan)}%0A` +
    `Sisa pinjaman: ${rupiah(sisa)}%0A%0A` +
    `Mohon segera melakukan pembayaran angsuran.%0A%0A` +
    `Terima kasih.`;

  const url = `https://wa.me/${nomor}?text=${pesan}`;
  window.open(url, "_blank");
}
